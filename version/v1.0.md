# V1.0 基础版开发计划 - 商城+进销存系统

## 📋 项目概述

### 版本目标
实现一个基础但完整的商城+进销存系统，支持：
- 商品管理（SPU/SKU）
- 基础商城功能（浏览、购物车、下单）
- 进销存核心流程（采购、销售、库存）
- 多仓库库存管理
- 基础财务报表

### 技术栈确认
| 层级 | 技术 |
|------|------|
| 后端 | NestJS + TypeScript + Prisma |
| 前端 | Vue3 + TypeScript + Naive UI |
| 数据库 | PostgreSQL |
| 缓存 | Redis |
| 文件存储 | 本地/MinIO（后续可换OSS） |

// 商品SPU（标准产品单元）
model Product {
  id           Int       @id @default(autoincrement())
  name         String    // 商品名称
  spuCode      String    @unique // SPU编码
  categoryId   Int       // 分类ID
  category     Category  @relation(fields: [categoryId], references: [id])
  brandId      Int?      // 品牌ID
  brand        Brand?    @relation(fields: [brandId], references: [id])
  description  String?   // 商品描述
  detail       String?   @db.Text // 商品详情（富文本）
  mainImage    String?   // 主图
  images       String[]  // 图片列表
  unitId       Int       // 单位ID
  unit         Unit      @relation(fields: [unitId], references: [id])
  
  // 规格属性（如：颜色、尺码）
  specTemplate Json?     // 规格模板 [{name:"颜色",values:["红","蓝"]}]
  
  // 状态
  status       ProductStatus @default(DRAFT)
  isEnabled    Boolean   @default(true)
  
  // 关联
  skus         ProductSku[]
  
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt
}

// 商品SKU（库存单元）
model ProductSku {
  id          Int       @id @default(autoincrement())
  skuCode     String    @unique // SKU编码
  productId   Int       // SPU ID
  product     Product   @relation(fields: [productId], references: [id], onDelete: Cascade)
  
  // 规格组合（如：{"颜色": "红", "尺码": "XL"}）
  specs       Json      
  
  // 价格
  costPrice   Decimal   @db.Decimal(12, 2) // 成本价
  salePrice   Decimal   @db.Decimal(12, 2) // 销售价
  marketPrice Decimal?  @db.Decimal(12, 2) // 市场价
  
  // 图片
  image       String?   // SKU图片
  
  // 条码
  barcode     String?   @unique // 条形码
  weight      Decimal?  @db.Decimal(10, 3) // 重量(kg)
  volume      Decimal?  @db.Decimal(10, 3) // 体积(m³)
  
  // 库存（各仓库）
  inventories Inventory[]
  
  // 商城显示
  isDefault   Boolean   @default(false) // 是否默认SKU
  sort        Int       @default(0)
  status      SkuStatus @default(ACTIVE)
  
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
}

// 库存记录（按仓库+SKU）
model Inventory {
  id          Int       @id @default(autoincrement())
  skuId       Int
  sku         ProductSku @relation(fields: [skuId], references: [id], onDelete: Cascade)
  warehouseId Int
  warehouse   Warehouse  @relation(fields: [warehouseId], references: [id])
  
  // 库存数量
  quantity    Int       @default(0) // 实际库存
  locked      Int       @default(0) // 锁定库存（已下单未发货）
  available   Int       @default(0) // 可用库存 = quantity - locked
  
  // 预警
  minStock    Int       @default(0) // 安全库存下限
  maxStock    Int       @default(999999) // 库存上限
  
  // 库位（可选）
  location    String?   // 库位编码
  
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt
  
  @@unique([skuId, warehouseId])
}

enum ProductStatus {
  DRAFT     // 草稿
  PENDING   // 待审核
  APPROVED  // 已通过
  REJECTED  // 已拒绝
  DISABLED  // 已禁用
}

enum SkuStatus {
  ACTIVE    // 启用
  INACTIVE  // 禁用
  DELETED   // 已删除
}
```

// 采购订单
model Purchase {
  id           Int            @id @default(autoincrement())
  orderNo      String         @unique // 采购单号 CG202402010001
  supplierId   Int
  supplier     Supplier       @relation(fields: [supplierId], references: [id])
  warehouseId  Int            // 入库仓库
  
  // 金额
  totalAmount  Decimal        @db.Decimal(12, 2) // 总金额
  discount     Decimal        @default(0) @db.Decimal(12, 2) // 折扣
  payable      Decimal        @db.Decimal(12, 2) // 应付金额
  paid         Decimal        @default(0) @db.Decimal(12, 2) // 已付金额
  
  // 状态
  status       PurchaseStatus @default(PENDING)
  
  // 时间
  orderDate    DateTime       @default(now()) // 下单日期
  deliveryDate DateTime?      // 预计到货日期
  
  // 关联
  items        PurchaseItem[]
  receipts     PurchaseReceipt[]
  payments     Payment[]
  
  remark       String?
  createdBy    Int
  createdAt    DateTime       @default(now())
  updatedAt    DateTime       @updatedAt
}

model PurchaseItem {
  id         Int      @id @default(autoincrement())
  purchaseId Int
  purchase   Purchase @relation(fields: [purchaseId], references: [id], onDelete: Cascade)
  skuId      Int
  sku        ProductSku @relation(fields: [skuId], references: [id])
  
  quantity   Int      // 采购数量
  received   Int      @default(0) // 已入库数量
  price      Decimal  @db.Decimal(12, 2) // 单价
  amount     Decimal  @db.Decimal(12, 2) // 金额
  
  createdAt  DateTime @default(now())
}

// 采购入库单
model PurchaseReceipt {
  id          Int              @id @default(autoincrement())
  receiptNo   String           @unique // 入库单号 RK202402010001
  purchaseId  Int
  purchase    Purchase         @relation(fields: [purchaseId], references: [id])
  warehouseId Int
  
  status      ReceiptStatus    @default(PENDING)
  
  totalAmount Decimal          @db.Decimal(12, 2)
  items       PurchaseReceiptItem[]
  
  remark      String?
  createdBy   Int
  createdAt   DateTime         @default(now())
  updatedAt   DateTime         @updatedAt
}

model PurchaseReceiptItem {
  id        Int             @id @default(autoincrement())
  receiptId Int
  receipt   PurchaseReceipt @relation(fields: [receiptId], references: [id], onDelete: Cascade)
  skuId     Int
  sku       ProductSku      @relation(fields: [skuId], references: [id])
  
  quantity  Int             // 入库数量
  price     Decimal         @db.Decimal(12, 2) // 入库单价（用于计算成本）
  
  createdAt DateTime        @default(now())
}

enum PurchaseStatus {
  PENDING    // 待审核
  APPROVED   // 已审核
  PARTIAL    // 部分入库
  COMPLETED  // 已完成
  CANCELLED  // 已取消
}

enum ReceiptStatus {
  PENDING   // 待入库
  RECEIVED  // 已入库
  CANCELLED // 已取消
}
```

// 库存流水（所有库存变动记录）
model InventoryLog {
  id          Int            @id @default(autoincrement())
  type        InventoryType  // 变动类型
  skuId       Int
  sku         ProductSku     @relation(fields: [skuId], references: [id])
  warehouseId Int
  
  quantity    Int            // 变动数量（正数入库，负数出库）
  before      Int            // 变动前库存
  after       Int            // 变动后库存
  
  // 关联业务
  bizType     String?        // 业务类型：PURCHASE/SALE/TRANSFER/ADJUST
  bizId       Int?           // 业务单据ID
  bizNo       String?        // 业务单据号

// 库存调拨单
model Transfer {
  id          Int            @id @default(autoincrement())

## 🔧 技术实现要点

### 1. 库存扣减策略
```typescript
// 使用数据库事务 + 乐观锁防止超卖
// 伪代码：
prisma.$transaction(async (tx) => {
  // 1. 查询库存并锁定
  const inventory = await tx.inventory.findUnique({
    where: { id: skuId },
    select: { available: true }
  });
  
  // 2. 校验库存
  if (inventory.available < quantity) {
    throw new Error('库存不足');
  }
  
  // 3. 扣减可用库存，增加锁定库存
  await tx.inventory.update({
    where: { id: skuId },
    data: {
      available: { decrement: quantity },
      locked: { increment: quantity }
    }
  });
  
  // 4. 创建订单...
});
```

### 2. 单号生成规则
| 单据类型 | 前缀 | 示例 |
|----------|------|------|
| 采购订单 | CG | CG202402010001 |
| 采购入库 | RK | RK202402010001 |
| 销售订单 | XS | XS202402010001 |
| 商城订单 | DD | DD202402010001 |
| 发货单 | FH | FH202402010001 |
| 调拨单 | DB | DB202402010001 |
| 调整单 | TZ | TZ202402010001 |

格式：`前缀 + 年月日 + 4位流水号`

### 3. 订单状态流转
```
商城订单：
待付款 → 已取消（超时未付）
待付款 → 已付款 → 待发货 → 已发货 → 已收货 → 已完成
                    ↓
              申请退款 → 退款中 → 已退款

销售订单（后台）：
待处理 → 已确认 → 待发货 → 已发货 → 已收货 → 已完成
   ↓
已取消
```

### 4. 成本核算（V1.0简化版）
- 使用**移动加权平均法**
- 每次入库时重新计算成本价
- 公式：`新成本价 = (原库存金额 + 本次入库金额) / (原库存数量 + 本次入库数量)`
